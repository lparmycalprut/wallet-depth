# Runtime dependencies for Streamlit Community Cloud.
#
# Streamlit 1.61.0 is incompatible with Starlette 1.4.0: the latter made
# GZipResponder.thread_minimum_size mandatory, while Streamlit's gzip wrapper
# does not pass it. Every browser request then returns HTTP 500, which appears
# as a page that spins forever.
#
# Keep this pair exact rather than a range. It makes Cloud redeploys
# reproducible and prevents its resolver from selecting the broken 1.61.0 /
# 1.4.0 combination. Streamlit 1.61.1 explicitly supports this Starlette
# version, and Starlette 1.3.1 retains the compatible GZipResponder signature.
streamlit==1.61.1
starlette==1.3.1

pandas
plotly
base58
curl_cffi
# Required by share_card.py for PNG share-card generation.
Pillow>=10.0
